export const rateLimiter = async (req, res, next) => {
  const clientKey = `ip:${req.ip}`;
  const requests = await redis.incr(clientKey);
  if (requests === 1) await redis.expire(clientKey, 60);
  if (requests > 100) return res.status(429).json({ error: "Rate limit exceeded" });
  next();
};
